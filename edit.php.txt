<?php
// ============================================
// edit.php — Form Edit Data Kas
// ============================================
require_once 'koneksi.php';

$id = isset($_GET['id']) ? (int)$_GET['id'] : 0;
if (!$id) { header("Location: index.php"); exit; }

// Ambil data lama
$row = $conn->query("SELECT * FROM kas WHERE id = $id")->fetch_assoc();
if (!$row) { header("Location: index.php"); exit; }

$error = [];
$nis     = $row['nis'];
$nama    = $row['nama_siswa'];
$nominal = $row['nominal'];
$tanggal = $row['tanggal_setor'];

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $nis      = trim($_POST['nis'] ?? '');
    $nama     = trim($_POST['nama_siswa'] ?? '');
    $nominal  = trim($_POST['nominal'] ?? '');
    $tanggal  = trim($_POST['tanggal_setor'] ?? '');

    if (empty($nis))     $error[] = 'NIS wajib diisi.';
    if (empty($nama))    $error[] = 'Nama siswa wajib diisi.';
    if (empty($nominal) || !is_numeric($nominal) || $nominal <= 0)
                         $error[] = 'Nominal harus berupa angka positif.';
    if (empty($tanggal)) $error[] = 'Tanggal setor wajib diisi.';

    if (empty($error)) {
        $stmt = $conn->prepare("UPDATE kas SET nis=?, nama_siswa=?, nominal=?, tanggal_setor=? WHERE id=?");
        $stmt->bind_param("ssdsi", $nis, $nama, $nominal, $tanggal, $id);
        $stmt->execute();
        $stmt->close();
        header("Location: index.php?pesan=ubah");
        exit;
    }
}
?>
<!DOCTYPE html>
<html lang="id">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Edit Data Kas</title>
    <link rel="stylesheet" href="style.css">
</head>
<body>

<div class="header">
    <div>
        <h1>✏️ Edit Data Kas</h1>
        <div class="subtitle">Perbarui data setoran kas siswa</div>
    </div>
    <a href="index.php" class="btn btn-secondary">← Kembali</a>
</div>

<!-- Error -->
<?php if (!empty($error)): ?>
    <?php foreach ($error as $e): ?>
        <div class="alert alert-danger">⚠️ <?= htmlspecialchars($e) ?></div>
    <?php endforeach; ?>
<?php endif; ?>

<div class="container">
    <div class="container-header">📝 Edit Data &mdash; <?= htmlspecialchars($row['nama_siswa']) ?></div>
    <div class="container-body">
        <form method="POST">
            <div class="form-grid">
                <div class="form-group">
                    <label for="nis">NIS (Nomor Induk Siswa)</label>
                    <input type="text" id="nis" name="nis" value="<?= htmlspecialchars($nis) ?>" required>
                </div>
                <div class="form-group">
                    <label for="nama_siswa">Nama Siswa</label>
                    <input type="text" id="nama_siswa" name="nama_siswa" value="<?= htmlspecialchars($nama) ?>" required>
                </div>
                <div class="form-group">
                    <label for="nominal">Nominal (Rp)</label>
                    <input type="number" id="nominal" name="nominal" value="<?= htmlspecialchars($nominal) ?>" min="1" required>
                </div>
                <div class="form-group">
                    <label for="tanggal_setor">Tanggal Setor</label>
                    <input type="date" id="tanggal_setor" name="tanggal_setor" value="<?= htmlspecialchars($tanggal) ?>" required>
                </div>
            </div>
            <div class="form-actions">
                <button type="submit" class="btn btn-warning">💾 Update Data</button>
                <a href="index.php" class="btn btn-secondary">✖ Batal</a>
            </div>
        </form>
    </div>
</div>
